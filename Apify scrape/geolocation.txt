BUFFER_M = 3000                        # 3 km buffer
CONTACT_EMAIL = "olivier@magicworx.be" # <-- jouw echte e-mail
# -------------------------------------------------------------------

HEADERS = {"User-Agent": f"magicworx-geo/1.0 ({CONTACT_EMAIL})"}
to_l72 = Transformer.from_crs(4326, 31370, always_xy=True).transform
to_wgs = Transformer.from_crs(31370, 4326, always_xy=True).transform

def boundary(town):
    r = requests.get("https://nominatim.openstreetmap.org/search",
        params={"q": f"{town}, Belgium","format":"json","polygon_geojson":1,"limit":5},
        headers=HEADERS, timeout=30)
    r.raise_for_status()
    for x in r.json():
        g = x.get("geojson", {})
        if g.get("type") in ("Polygon","MultiPolygon") and x.get("class")=="boundary":
            return shape(g), x.get("display_name")
    for x in r.json():
        g = x.get("geojson", {})
        if g.get("type") in ("Polygon","MultiPolygon"):
            return shape(g), x.get("display_name")
    return None, None

rows, problems, counts = {}, [], {}
print("========== CONTROLE (elke gemeente moet 'ok' zijn) ==========")
for regio, towns in REGIONS.items():
    print(f"\n--- {regio} ---")
    polys = []
    for t in towns:
        geom, name = boundary(t)
        if geom:
            print(f"  ok    {t:<24} -> {name}")
            polys.append(geom)
        else:
            print(f"  MISS  {t:<24} -> NIET GEVONDEN")
            problems.append((regio, t))
        time.sleep(1.1)
    counts[regio] = (len(polys), len(towns))
    if not polys:                               # beveiliging: geen enkele gemeente gevonden
        print(f"  !! GEEN gemeenten gevonden voor {regio} — overgeslagen")
        continue
    merged = unary_union(polys)
    buffered = transform(to_wgs, transform(to_l72, merged).buffer(BUFFER_M))
    buffered = buffered.simplify(0.0005, preserve_topology=True)
    rows[regio] = (", ".join(towns), json.dumps(mapping(buffered)))
    print(f"  => {len(polys)}/{len(towns)} gemeenten gevonden, {len(rows[regio][1])} tekens")

print("\n\n#####  PLAK-BLOK — kopieer alles hieronder, plak in cel A2  #####")
print("#####  (kolom A=regio, B=gemeenten, C=geolocation)          #####\n")
for regio, (gem, poly) in rows.items():
    print(f"{regio}\t{gem}\t{poly}")

print("\n\n========== SAMENVATTING (scan deze eerst) ==========")
for regio, (found, exp) in counts.items():
    length = len(rows[regio][1]) if regio in rows else 0
    flags = []
    if found != exp:      flags.append("GEMEENTE ONTBREEKT")
    if length > 45000:    flags.append("TE LANG — verhoog simplify naar 0.001")
    if found == 0:        flags.append("LEEG")
    flag = ("   <-- " + ", ".join(flags)) if flags else ""
    print(f"  {regio:<24} {found}/{exp} gem   {length:>6} tekens{flag}")

if problems:
